/**
 * 설문 데이터를 기반으로 질문 이름과 선택지 텍스트를 매핑하는 객체를 생성합니다.
 * @returns {object} 질문 이름과 선택지 맵을 담은 객체
 */
let questionTextMap = null;
function getQuestionTextMap() {
    if (questionTextMap) {
        return questionTextMap;
    }
    const surveyData = getSurveyData();
    const map = {};
    surveyData.parts.forEach(part => {
        part.questions.forEach(q => {
            if (q.options) {
                map[q.name] = {};
                q.options.forEach((opt, index) => {
                    const code = index + 1;
                    map[q.name][code.toString()] = opt;
                });
            }
        });
    });
    questionTextMap = map;
    return questionTextMap;
}

// ===============================================================
// === 3단계: 웹 앱 백엔드 - 응답을 즉시 대기열에 저장 (최적화 버전) ===
// ===============================================================
function processForm(formObject) {
  try {
    const sheetId = PropertiesService.getScriptProperties().getProperty('spreadsheetId');
    if (!sheetId) {
      Logger.log("오류: setup 함수가 실행되지 않아 sheetId를 찾을 수 없습니다.");
      return { status: "error", message: "설문 시스템이 초기화되지 않았습니다. 관리자에게 문의하세요." };
    }
    
    const ss = SpreadsheetApp.openById(sheetId);
    let queueSheet = findActiveQueueSheet(ss);
    if (!queueSheet) {
        queueSheet = ss.getSheetByName("대기열 (Queue)"); // fall back to original name
        if (!queueSheet) return { status: "error", message: "대기열 시트를 찾을 수 없습니다." };
    }

    const stringifiedData = JSON.stringify(formObject);
    const rowData = [new Date(), stringifiedData];
    
    // safeAppendRow 함수를 사용하여 Sheets API 충돌 시 자동 재시도 (최대 6회)
    const success = safeAppendRow(queueSheet, rowData, 6); 

    if (success) {
      return { status: "success" };
    } else {
      // 6회 재시도 후에도 최종 실패한 경우 (Google 부하가 극심한 경우)
      return { status: "error", message: "서버 부하로 인해 요청 처리에 실패했습니다. 잠시 후 다시 시도해 주세요." };
    }

  } catch (error) {
    Logger.log("!!! processForm에서 오류 발생: " + error.toString());
    return { status: "error", message: "서버 오류 발생: " + error.message };
  }
}

/**
 * [UTILITY FUNCTION] 시트에 appendRow를 안전하게 시도하는 함수 (지수적 백오프 + 지터 + 실패 로그 포함)
 */
function safeAppendRow(sheet, data, maxRetries) {
  const ss = sheet.getParent();
  const logSheet = ss.getSheetByName("대기열 오류 로그"); // setup()에서 이미 생성됨
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      sheet.appendRow(data);
      return true; // 성공하면 즉시 종료
    } catch (err) {
      // 마지막 시도이면 로그에 기록
      if (i === maxRetries - 1) {
        const errRow = [new Date(), "appendRow 실패", err.toString(), JSON.stringify(data)];
        try { logSheet.appendRow(errRow); } catch(e) { Logger.log("로그 쓰기 실패: " + e); }
        return false;
      }
      
      // 지수적 백오프 (Exponential Backoff) + 지터 (Jitter)
      const base = 200; // 기본 대기(ms)
      const backoff = base * Math.pow(2, i); // 200, 400, 800, 1600...
      const jitter = Math.floor(Math.random() * 200); // 0~199ms의 랜덤 대기 시간 추가
      Utilities.sleep(backoff + jitter);
    }
  }
  return false;
}

// ===============================================================
// === 4단계: 대기열 처리 (트리거로 실행) - 운영 완성판 ===
// ===============================================================
const QUEUE_SHEET_MAX_ROWS = 1000; // 큐 시트 성능 저하를 막기 위한 최대 행 수
const BATCH_LIMIT = 200; // 한 번에 처리할 행 수

function processQueue() {
  const lock = LockService.getScriptLock();
  
  // 큐 소비자가 동시에 실행되는 것을 방지하기 위해 락을 시도합니다. (최대 30초 대기)
  try {
    if (!lock.tryLock(30000)) {
      Logger.log("processQueue: 다른 인스턴스가 실행 중입니다. (30초 대기 실패)");
      return;
    }

    const sheetId = PropertiesService.getScriptProperties().getProperty('spreadsheetId');
    if (!sheetId) return;
    
    const ss = SpreadsheetApp.openById(sheetId);
    let queueSheet = findActiveQueueSheet(ss);
    
    if (!queueSheet) {
        Logger.log('활성 대기열 시트를 찾을 수 없습니다.');
        return;
    }

    const responseSheet = ss.getSheetByName("응답");
    const errorLog = ss.getSheetByName("대기열 처리 오류");
    const statsSheet = ss.getSheetByName("운영 통계 로그");
    
    const lastRow = queueSheet.getLastRow();
    
    // 1행은 헤더(제목) 행이라고 가정
    if (lastRow <= 1) {
        recordStats(statsSheet, 0, lastRow);
        return;
    }
    
    let processedCount = 0;
    
    const headers = responseSheet.getRange(1, 1, 1, responseSheet.getLastColumn()).getValues()[0];
    const questionMap = getQuestionTextMap();

    const finalRowToProcess = lastRow;

    for (let start = 2; start <= finalRowToProcess; start += BATCH_LIMIT) {
      const chunkSize = Math.min(BATCH_LIMIT, finalRowToProcess - start + 1);
      
      const queueData = queueSheet.getRange(start, 1, chunkSize, 2).getValues();

      const rowsToAdd = [];
      let currentQueueRowIndex = start;

      // 1. 안전하게 각 행 파싱 및 매핑
      for (let i = 0; i < queueData.length; i++) {
        const row = queueData[i];
        let formObject;
        
        try {
          formObject = JSON.parse(row[1]);
        } catch (err) {
          errorLog.appendRow([new Date(), "JSON Parse 실패", currentQueueRowIndex, row[1], err.toString()]);
          currentQueueRowIndex++;
          continue;
        }

        const timestamp = row[0];
        const mapped = headers.map(header => {
          if (header === "타임스탬프") return timestamp;

          const value = formObject[header];
          if (value === undefined || value === null || value === "") return "";

          // '기타' 필드 처리
          if (questionMap[header] && value === '99') {
            return formObject[header + '_기타'] || "";
          } 
          
          // 일반 선택지/텍스트 매핑
          if (questionMap[header] && questionMap[header][value]) {
            return questionMap[header][value];
          } 
          
          // 서술형, 연락처, 또는 기타 텍스트 필드
          return formObject[header] || value || ""; 
        });
        
        rowsToAdd.push(mapped);
        currentQueueRowIndex++;
      }

      if (rowsToAdd.length > 0) {
        // 2. 응답 시트에 배치 쓰기 (Batch Write)
        responseSheet.getRange(responseSheet.getLastRow() + 1, 1, rowsToAdd.length, headers.length).setValues(rowsToAdd);
      }
      
      // 3. 삭제: 청크만큼 안전하게 삭제 
      try {
        queueSheet.deleteRows(start, rowsToAdd.length); 
      } catch (e) {
        errorLog.appendRow([new Date(), "deleteRows 실패", start, rowsToAdd.length, e.toString()]);
      }

      processedCount += rowsToAdd.length;
    }

    // 4. 시트 용량 관리: 1000행 초과시 시트 자동 분할
    checkAndRollQueueSheet(ss, queueSheet);

    // 5. 통계 기록
    recordStats(statsSheet, processedCount, queueSheet.getLastRow());

  } catch (error) {
    Logger.log("❌ Error in processQueue: " + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) { /* lock was not acquired or already released */ }
  }
}

/**
 * [UTILITY FUNCTION] 활성 (가장 최신) 대기열 시트를 찾습니다.
 */
function findActiveQueueSheet(ss) {
    const sheets = ss.getSheets();
    const queueSheets = sheets.filter(s => s.getName().startsWith("대기열 (Queue)"));
    
    if (queueSheets.length === 0) return null;

    // 인덱스를 기준으로 정렬하여 가장 최신 시트를 찾습니다.
    queueSheets.sort((a, b) => {
        const indexA = parseInt(a.getName().match(/\((\d+)\)/)?.[1] || 1);
        const indexB = parseInt(b.getName().match(/\((\d+)\)/)?.[1] || 1);
        
        // ARCHIVE 시트는 가장 뒤로 보냅니다.
        const isArchiveA = a.getName().includes("ARCHIVE");
        const isArchiveB = b.getName().includes("ARCHIVE");

        if (isArchiveA && !isArchiveB) return 1;
        if (!isArchiveA && isArchiveB) return -1;
        
        return indexB - indexA; // 내림차순 정렬 (가장 큰 인덱스가 맨 위)
    });
    
    return queueSheets[0];
}

/**
 * [UTILITY FUNCTION] 대기열 시트 행 수가 초과되면 새로운 시트를 생성하고 큐를 롤오버합니다.
 */
function checkAndRollQueueSheet(ss, currentQueueSheet) {
    if (currentQueueSheet.getLastRow() > QUEUE_SHEET_MAX_ROWS) {
        let newIndex = 1;
        
        const currentName = currentQueueSheet.getName();
        const match = currentName.match(/\((\d+)\)/);
        
        if (match) {
            newIndex = parseInt(match[1]) + 1;
            // 현재 시트를 ARCHIVE로 변경
            currentQueueSheet.setName(`대기열 (Queue) - ${match[1]}-ARCHIVE-${new Date().getFullYear()}`);
        } else {
             // 최초 시트의 이름 변경
             currentQueueSheet.setName(`대기열 (Queue) - 1-ARCHIVE-${new Date().getFullYear()}`);
             newIndex = 2;
        }

        // 새로운 활성 Queue 시트 생성
        const newQueueSheet = ss.insertSheet(`대기열 (Queue) (${newIndex})`);
        const queueHeaders = ["타임스탬프 (요청)", "응답 데이터 (JSON)"];
        newQueueSheet.getRange(1, 1, 1, queueHeaders.length).setValues([queueHeaders]);
        newQueueSheet.setFrozenRows(1);
        Logger.log(`✅ 대기열 시트가 ${QUEUE_SHEET_MAX_ROWS}행을 초과하여 새로운 시트 '대기열 (Queue) (${newIndex})'로 자동 분할되었습니다.`);
    }
}

/**
 * [UTILITY FUNCTION] 처리 결과를 통계 로그 시트에 기록합니다.
 */
function recordStats(sheet, processedCount, queueSize) {
    if (sheet) sheet.appendRow([new Date(), processedCount, queueSize]);
}

/**
 * [UTILITY FUNCTION] 대기열 오류 로그 시트에 기록된 실패 항목들을 재시도합니다.
 * 이 함수는 별도의 트리거 (예: 1시간 주기)로 실행되도록 설정하는 것을 권장합니다.
 */
function retryFailedQueueWrites() {
  const sheetId = PropertiesService.getScriptProperties().getProperty('spreadsheetId');
  if (!sheetId) return;
  const ss = SpreadsheetApp.openById(sheetId);
  const errorLogSheet = ss.getSheetByName("대기열 오류 로그");
  const queueSheet = findActiveQueueSheet(ss); // 현재 활성 큐 시트에 재기록 시도

  if (!errorLogSheet || !queueSheet || errorLogSheet.getLastRow() <= 1) return;

  const lastRow = errorLogSheet.getLastRow();
  const range = errorLogSheet.getRange(2, 1, lastRow - 1, 4);
  const failedData = range.getValues();
  
  const successfulRetries = [];
  
  // JSON 데이터만 추출하여 재시도 리스트 구성
  for (let i = 0; i < failedData.length; i++) {
    const row = failedData[i];
    const dataString = row[3]; // 4번째 열(인덱스 3)에 JSON 문자열이 저장되어 있습니다.
    
    try {
        const dataArray = [new Date(), dataString];
        // 안전한 쓰기 함수를 사용하여 재시도. 성공 시 큐에 다시 들어갑니다.
        const success = safeAppendRow(queueSheet, dataArray, 3); // 재시도 3회
        
        if (success) {
            successfulRetries.push(i + 2); // 성공한 행 번호 (헤더 제외)
        }
    } catch (e) {
        Logger.log(`Retry failed for row ${i + 2} (Critically): ${e}`);
    }
  }

  // 성공적으로 재시도되어 큐에 재기록된 행 삭제
  if (successfulRetries.length > 0) {
    // 삭제는 역순으로 해야 안전함
    successfulRetries.sort((a, b) => b - a);
    successfulRetries.forEach(rowNum => {
      errorLogSheet.deleteRow(rowNum);
    });
    Logger.log(`✅ ${successfulRetries.length}개의 실패 항목이 대기열에 재기록되었습니다.`);
  }
}
