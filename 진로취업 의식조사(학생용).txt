let questionTextMap = null;
function getQuestionTextMap() {
    if (questionTextMap) return questionTextMap;
    const surveyData = getSurveyData();
    const map = {};
    surveyData.parts.forEach(part => {
        part.questions.forEach(q => {
            if (q.options) {
                map[q.name] = {};
                q.options.forEach((opt, index) => {
                    const cleanOpt = opt.replace(/<[^>]*>/g, ""); 
                    map[q.name][(index + 1).toString()] = cleanOpt;
                });
            }
        });
    });
    questionTextMap = map;
    return map;
}

 * 폼 데이터를 처리하고 Queue에 저장합니다.
 */
function processForm(formObject) {
  try {
    const sheetId = PropertiesService.getScriptProperties().getProperty('spreadsheetId');
    if (!sheetId) return { status: "error", message: "서버 오류: Spreadsheet ID가 설정되지 않았습니다." };

    // 🔑 UUID 생성 및 삽입
    formObject.uuid = Utilities.getUuid(); 
    
    const phone = cleanPhoneNumber(formObject['연락처']);
    
    if (!phone || !/^\d{10,11}$/.test(phone)) {
        return { status: "error", message: '올바른 형식의 연락처를 입력해주세요. (예: 01012345678)' };
    }

    const ss = SpreadsheetApp.openById(sheetId);
    const responseSheet = ss.getSheetByName(SHEET_NAME_RESPONSE);
    const headers = getHeaders(responseSheet);
    
    // 1단계 방어: 제출 즉시 중복 확인 (Cache Lock)
    const cache = CacheService.getScriptCache();
    const cacheKeyPhone = 'phone_' + phone;
    
    if (cache.get(cacheKeyPhone)) {
      return { status: "error", message: '이미 제출되었거나 처리 중입니다. 잠시 후 다시 시도해 주세요.' };
    }
    cache.put(cacheKeyPhone, 'submitted', CACHE_LOCK_TIMEOUT); // 2분간 임시 잠금

    // 💡 2단계 방어 (강화): 최종 DB 중복 체크 - Queue 삽입 전에 Lock 걸고 강제 검사
    const scriptLock = LockService.getScriptLock();
    let isDuplicate = false;

    // Lock 획득 시도 (Queue가 비어있는 상태에서 DB 조회 시 동시성 보장)
    if (scriptLock.tryLock(LOCK_TIMEOUT)) {
        try {
            const { existingPhones } = getExistingPhonesAndUUIDs(responseSheet, headers);
            
            if (existingPhones.has(phone)) {
                isDuplicate = true;
            }
        } catch (e) {
            Logger.log("processForm DB 조회 중 오류 발생 (Lock 내부): " + e.toString());
            // DB 조회 실패 시, 잠금 해제 후 바로 에러 반환하여 Queue 삽입 차단
            scriptLock.releaseLock();
            return { status: "error", message: "시스템 오류로 중복 응답 확인 실패. 잠시 후 다시 시도해주세요." };
        } finally {
            scriptLock.releaseLock();
        }
    } else {
        // ❗ Lock 획득 실패 시, 중복 검사 자체가 불가능하므로 안전을 위해 Queue 삽입을 차단합니다.
        Logger.log("processForm Lock 획득 실패: processQueue 실행 중. 안전을 위해 응답 차단.");
        return { status: "error", message: "시스템 처리 중입니다. 잠시 후 다시 시도해주세요." };
    }
    
    // 최종 중복 검사 결과 처리
    if (isDuplicate) {
        Logger.log(`processForm 단계에서 최종 DB 중복 응답 차단: ${phone}`);
        return { status: "error", message: "이미 응답이 제출되었습니다." };
    }


    let queueSheet = ss.getSheetByName(SHEET_NAME_QUEUE);
    if (!queueSheet) {
        queueSheet = ss.insertSheet(SHEET_NAME_QUEUE);
        queueSheet.appendRow(["타임스탬프 (요청)", "응답 데이터 (JSON)"]);
    }
    
    // 큐에 데이터 삽입
    safeAppendRow(queueSheet, [new Date(), JSON.stringify(formObject)], 5); 
    
    // 큐 삽입 성공 후 Cache Lock 해제 (processQueue에서 처리할 때 다시 Cache Lock을 사용할 것이므로 유지)
    // Cache Lock은 processQueue가 해제하지 않으므로, 2분 뒤 자동으로 풀림.

    return { status: "success" };

  } catch (error) {
    // 큐 삽입 실패 시 Cache 잠금 해제
    if (formObject && formObject['연락처']) {
        const phone = cleanPhoneNumber(formObject['연락처']);
        const cache = CacheService.getScriptCache();
        cache.remove('phone_' + phone); 
    }
    Logger.log("processForm 치명적 오류: " + error.toString());
    return { status: "error", message: "서버 오류 발생: " + error.message };
  }
}


/**
 * 1분마다 실행되는 함수. 대기열을 처리하고 DB에 데이터를 추가합니다.
 */
function processQueue() {
  const sheetId = PropertiesService.getScriptProperties().getProperty('spreadsheetId');
  if (!sheetId) return;

  const ss = SpreadsheetApp.openById(sheetId);
  const queueSheet = ss.getSheetByName(SHEET_NAME_QUEUE);
  const statsSheet = ss.getSheetByName("운영 통계 로그");

  if (!queueSheet || queueSheet.getLastRow() <= 1) {
      if(statsSheet) recordStats(statsSheet, 0, 0);
      return;
  }

  const responseSheet = ss.getSheetByName(SHEET_NAME_RESPONSE);
  const headers = getHeaders(responseSheet);
  const questionMap = getQuestionTextMap();

  const numRowsToProcess = Math.min(BATCH_LIMIT, queueSheet.getLastRow() - 1);
  if (numRowsToProcess <= 0) return;

  // 1. Lock 획득 (DB Read, DB Write, Queue Delete 전체를 묶음)
  const scriptLock = LockService.getScriptLock();
  
  // 🔑 Lock 경합 보완: 5초 대기 후 재시도
  if (!scriptLock.tryLock(LOCK_TIMEOUT)) {
      Utilities.sleep(RETRY_WAIT_TIME); // 5초 대기
      if (!scriptLock.tryLock(LOCK_TIMEOUT)) {
        Logger.log("LOCK 실패 (재시도 후): processQueue가 잠금을 얻지 못했습니다.");
        if(statsSheet) recordStats(statsSheet, 0, queueSheet.getLastRow() -1);
        return; // 최종 실패
      }
      Logger.log("LOCK 성공 (5초 대기 후): 잠금 획득");
  }

  let queueData = null;
  let result = { rowsToAdd: [], numProcessed: 0 };
  let dbWriteSuccess = false; // DB 저장 성공 여부 플래그

  try {
    // 큐 데이터 읽기
    queueData = queueSheet.getRange(2, 1, numRowsToProcess, 2).getValues();
    
    // 2. 큐 처리 및 중복 검사 (DB Read 포함)
    result = checkExistingPhonesAndGetRows(ss, queueData, headers, questionMap);
    
    // 3. 응답 데이터 저장 (DB Write) 
    if (result.rowsToAdd.length > 0) {
        responseSheet.getRange(responseSheet.getLastRow() + 1, 1, result.rowsToAdd.length, headers.length).setValues(result.rowsToAdd);
        dbWriteSuccess = true;
    }
    
    // 5. 통계 기록 (Lock 해제 전)
    if(statsSheet) recordStats(statsSheet, result.rowsToAdd.length, queueSheet.getLastRow() -1);

  } catch (error) {
    Logger.log("processQueue (Lock 내부) 치명적 오류: " + error.toString());
    const errorLog = ss.getSheetByName(SHEET_NAME_ERROR);
    safeAppendRow(errorLog, [new Date(), "processQueue 치명적 실패", "N/A", "N/A", error.toString()], 3);
  } finally {
    // 🔑 DB Write 성공 시에만 Queue 삭제 로직
    try {
        if (result.numProcessed > 0 && (dbWriteSuccess || result.rowsToAdd.length === 0)) {
            // 저장 성공(dbWriteSuccess) 또는 모두 중복 처리(rowsToAdd=0)된 경우에만 삭제
            queueSheet.deleteRows(2, result.numProcessed); 
        } else if (!dbWriteSuccess && result.rowsToAdd.length > 0) {
             // DB 저장이 실패했는데 rowsToAdd가 남아있는 경우: 데이터 유실 방지를 위해 Queue 삭제를 보류
             Logger.log("응답 시트 저장 실패 (rowsToAdd > 0), Queue 삭제 보류: 다음 실행 시 재시도");
        }
    } catch (e) {
        Logger.log("Queue 삭제 중 오류 발생: " + e.toString());
    }
    
    scriptLock.releaseLock();
  }
}


/**
 * 응답 시트의 기존 전화번호를 조회하고, 큐 데이터를 처리 가능한 형태로 매핑합니다.
 */
function checkExistingPhonesAndGetRows(ss, queueData, headers, questionMap) {
    const responseSheet = ss.getSheetByName(SHEET_NAME_RESPONSE);
    
    // 💡 새로운 함수를 사용하여 기존 데이터 조회 (중복 처리의 최종 방어선)
    const { existingPhones, existingUUIDs } = getExistingPhonesAndUUIDs(responseSheet, headers);

    const rowsToAdd = [];
    const processedPhonesInBatch = new Set();
    const errorLog = ss.getSheetByName("대기열 오류 로그");
    let numProcessed = 0; 
    
    // 큐 데이터 처리
    for (const row of queueData) {
        numProcessed++;
        let formObject;
        
        try {
            formObject = JSON.parse(row[1]);
            const currentPhone = cleanPhoneNumber(formObject['연락처']);
            const currentUUID = formObject.uuid;

            
            // 3단계 방어: 최종 중복 검사
            if (existingPhones.has(currentPhone)) {
                Logger.log(`[비즈니스 중복] 응답 건너뛰기 (전화번호 중복): ${currentPhone}`);
                continue; 
            }
            if (existingUUIDs.has(currentUUID)) { 
                Logger.log(`[기술적 중복] 응답 건너뛰기 (UUID 중복): ${currentUUID}`);
                continue; 
            }
            if (processedPhonesInBatch.has(currentPhone)) {
                Logger.log(`[배치 내 중복] 응답 건너뛰기 (현재 배치 내 전화번호 중복): ${currentPhone}`);
                continue; 
            }

            const mapped = headers.map(header => {
                if (header === "타임스탬프") return row[0];
                if (header === "UUID") return currentUUID; 
                
                const value = formObject[header];
                if (value === undefined || value === null || value === "") return "";
                if (Array.isArray(value)) {
                    const questionMap = getQuestionTextMap();
                    if (!questionMap[header]) return value.join(', ');
                    return value.map(code => questionMap[header][code] || code).join(', ');
                }
                if (getQuestionTextMap()[header]) {
                    if (value === '99') return formObject[header + '_기타'] || "";
                    return getQuestionTextMap()[header][value] || value;
                }
                return value;
            });

            rowsToAdd.push(mapped);
            
            processedPhonesInBatch.add(currentPhone);
            existingPhones.add(currentPhone);
            existingUUIDs.add(currentUUID);

        } catch (err) {
            safeAppendRow(ss.getSheetByName("대기열 오류 로그"), [new Date(), "Row Processing 실패", "", row[1], err.toString()], 3);
        }
    }
    
    return { rowsToAdd: rowsToAdd, numProcessed: numProcessed }; 
}

/**
 * 응답 시트(DB)에서 기존 전화번호와 UUID를 모두 조회하여 Set으로 반환합니다.
 */
function getExistingPhonesAndUUIDs(responseSheet, headers) {
    let existingPhones = new Set();
    let existingUUIDs = new Set(); 
    
    if (!responseSheet || responseSheet.getLastRow() <= 1) {
        return { existingPhones, existingUUIDs };
    }
    
    const phoneColumnIndex = headers.indexOf('연락처') + 1; // 1-based index 
    const uuidColumnIndex = headers.indexOf('UUID') + 1;     // 1-based index 
    const rangeStartRow = 2;
    const rangeNumRows = responseSheet.getLastRow() - 1;

    // 1. 전화번호 열 전체를 읽기
    if (phoneColumnIndex > 0) {
        const phoneValues = responseSheet.getRange(rangeStartRow, phoneColumnIndex, rangeNumRows, 1).getValues();
        existingPhones = new Set(phoneValues.flat().map(val => cleanPhoneNumber(val)));
    }

    // 2. UUID 열 전체를 읽기
    if (uuidColumnIndex > 0) {
        const uuidValues = responseSheet.getRange(rangeStartRow, uuidColumnIndex, rangeNumRows, 1).getValues();
        existingUUIDs = new Set(uuidValues.flat());
    }
    
    return { existingPhones, existingUUIDs };
}


/**
 * 전화번호에서 하이픈, 공백, 괄호 등을 제거하고 숫자만 남깁니다.
 */
function cleanPhoneNumber(phone) {
    if (typeof phone === 'number') {
        phone = phone.toString();
    }
    if (typeof phone !== 'string') return '';
    return phone.replace(/[^0-9]/g, '');
}

/**
 * 시트의 헤더(첫 행)를 배열로 가져옵니다.
 */
function getHeaders(sheet) {
    if (!sheet || sheet.getLastRow() < 1) return [];
    return sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
}

/**
 * 오류 로그 시트에 안전하게 행을 추가합니다.
 */
function safeAppendRow(sheet, rowData, maxRetries) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      sheet.appendRow(rowData);
      return;
    } catch (err) {
      if (i === maxRetries - 1) {
        Logger.log(`최종 실패: safeAppendRow failed after ${maxRetries} retries.`);
        throw err; // 최종 실패 시 에러 발생
      }
      Utilities.sleep(300 * Math.pow(2, i) + Math.floor(Math.random() * 200));
    }
  }
}

function recordStats(sheet, processedCount, queueSize) {
    sheet.appendRow([new Date(), processedCount, Math.max(0, queueSize)]);
}
